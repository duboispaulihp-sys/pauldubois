\begin{abstract}
This is the master document of the geometric-drift program: it collects Parts I--XIV into
one place and, unlike the earlier companion, carries the \emph{proof bodies} in full. The
program constructs Paul Dubois's geometric mechanism around the Navier--Stokes blow-up
question, reduces its macroscopic content to exact integer arithmetic at a fixed literal
scale, and machine-checks the resulting chain in Lean~4: ten self-contained files in
\texttt{formal/}, $112$ theorems, axiom profile $[\mathrm{propext},\mathrm{Quot.sound}]$
throughout (a single theorem, \texttt{compareMono}, additionally uses
$\mathrm{Classical.choice}$), \texttt{sorryAx} never. The proof sections below give
complete written derivations for: the two workhorse potential lemmas and the
$1000/999$ accounting identity (Part~XI), the exact critical point in closed rational form,
the sharp no-side squeeze at the three critical ratios with its sliding-window certificate
(Part~XII), the constructed integral singularity trajectories with per-window errors
bounded by $999$ (Part~XIII), and the adversarial error-tolerance bracket
$\mathsf{Em} < \mathsf{Ens}$ with machine-constructed counterexamples on the far side
(Part~XIV). Every number quoted here is an exact integer certificate checked by the Lean
kernel or by independent stdlib verifiers; the boundary of what is and is not proved is
stated precisely in Section~\ref{sec:integrity}.
\end{abstract}

\section{The geometric framework (Parts I--II)}\label{sec:framework}

\subsection{Self-similar profile}
With blow-up time $1$, set
\[
\sigma=\log\tfrac{1}{1-t},\qquad
y=\frac{x}{\sqrt{1-t}},\qquad
u(t,x)=(1-t)^{-1/2}\,U(\sigma,y),
\]
so that $U$ satisfies the forced profile equation
\[
\bigl(\partial_\sigma+\tfrac12+\tfrac12\,y\cdot\nabla-\Delta\bigr)U+(U\cdot\nabla)U
= -\nabla p,\qquad \operatorname{div}U=0 .
\]

\subsection{The cascade}
Windows have width $S=\log 2$ (scale ratio $\lambda=\tfrac12$ per stage); the window-$n$
reference packet $W_n$ reproduces under the discrete symmetry
$W_{n+1}(\sigma,y)=R\,W_n(\sigma-S,R^{-1}y)$. The per-window excess is
\[
\delta_n=(\text{closure ratio in window } n)-\varepsilon_{\mathrm{prof}},
\qquad \varepsilon_{\mathrm{prof}}=\tfrac1{30},
\qquad \varepsilon_{\mathrm{drift}}=\sum_{n\ge n_0}\delta_n,
\]
and the total misfit is $\varepsilon_0=\varepsilon_{\mathrm{prof}}+\varepsilon_{\mathrm{drift}}$.
The certificate tilt constant is $\sin^2\theta\le 5.51\,D^2$.

\subsection{The budget}
On the band
\[
\sigma\in(7.3303,\,7.8240)
=\bigl(4\log\tfrac{25}{4},\;4\bigl(\tfrac52\log 2+\log\tfrac54\bigr)\bigr)
\]
the drift must obey $\sum_n\delta_n\le D_{\max}(\sigma)$, where $D_{\max}$ is the root of
\[
\bigl(\tfrac14-5.51\,D^2\bigr)\sigma
=\log 4+2\log\frac{1+\varepsilon_{\mathrm{prof}}+D}{1-\varepsilon_{\mathrm{prof}}-D}.
\]

\section{The integer reduction: admissible trajectories and anchors}\label{sec:reduction}

The mechanism's macroscopic content is carried by two integer sequences: the response
$\dhat n$ and the source floor $\chat n$. The coupled recursion at the fixed literal scale
$D=30{,}000{,}000$ with damping $Q=30{,}000$ (ratio $1000:1$, feedback $1000/999$) is
\begin{equation}\label{eq:coupling}
30000000\cdot \dhat(n+1)=30000\cdot \dhat n+30000000\cdot \chat n .
\end{equation}
A \emph{no-side} (resp.\ \emph{yes-side}) \emph{admissible} trajectory has the launch at or
above (resp.\ at or below) its cap ($\dhat 0\ge/\le 10^6$), the source at or above (below)
its cap ($\chat 0\ge/\le 39000$), and the source decaying no faster than (at most) a
contraction ratio $a/b$ per window, in the exact coupling-scale form
$30000000\cdot\chat(n+1)\ge A\cdot\chat n$ with $A=a\cdot 30000000/b$
(integrality of the multiplier is exact: $b\mid 30000000$).

\section{The accounting core: four lemmas with complete proofs}\label{sec:core}

Throughout, $D=30000000$, $Q=30000$, and $\sumto f\,K=f\,0+f\,1+\dots+f\,(K-1)$ with
$\sumto f\,0=0$.

\begin{lemma}[prefix step; \texttt{sumToStep}, \texttt{sumToMono}]\label{lem:step}
For every sequence $f$ and every $K$, $\sumto f\,(K+1)=\sumto f\,K+f\,K$ (definitionally:
$=\!\mathrm{rfl}$). Consequently, if $f\,n\ge 0$ for all $n$ then
$\sumto f\,K\le \sumto f\,(K+1)$ for all $K$.
\end{lemma}
\begin{proof}
The step equation is the defining clause of \texttt{sumTo}. Monotonicity: by induction on
$K$; the base is $0\le f\,0$, and the step adds $\sumto f\,K\le \sumto f\,K+f\,K$ with
$f\,K\ge 0$.
\end{proof}

\begin{lemma}[response envelope; \texttt{geometricEnvelope}]\label{lem:envelope}
Let $0<X<D$. If $D\cdot\dhat(n+1)\le X\cdot\dhat n$ for all $n$ and $\dhat n\ge 0$ for
all $n$, then for every $K$,
\[
(D-X)\cdot\sumto \dhat\,K\;+\;D\cdot\dhat\,K\;\le\;D\cdot\dhat\,0 .
\]
In particular $\sumto \dhat\,K\le \dhat\,0\cdot D/(D-X)$ for all $K$: the whole cascade
releases at most $\dhat\,0$ times the envelope factor.
\end{lemma}
\begin{proof}
Induction on $K$. Base $K=0$: $0+D\cdot\dhat\,0\le D\cdot\dhat\,0$. Step: assume
$(D-X)\sumto\dhat\,K+D\,\dhat\,K\le D\,\dhat\,0$. Then, inserting the definition of
$\sumto\dhat\,(K+1)$ and the contractivity hypothesis at $n=K$,
\[
(D-X)\sumto\dhat\,(K+1)+D\,\dhat\,(K+1)
=(D-X)\sumto\dhat\,K+\underbrace{(D-X)\,\dhat\,K+D\,\dhat\,(K+1)}_{\le\;D\,\dhat\,K}
\le (D-X)\sumto\dhat\,K+D\,\dhat\,K\le D\,\dhat\,0,
\]
because $D\,\dhat\,(K+1)\le X\,\dhat\,K$ gives $(D-X)\,\dhat\,K+D\,\dhat\,(K+1)\le D\,\dhat\,K$.
\end{proof}

\begin{lemma}[source potential; \texttt{chatBound1..3}, \texttt{chatUpper}, \texttt{chatLowerBound}]\label{lem:source}
Let $0<X'<D'$. If $D'\cdot\chat(n+1)\le X'\cdot\chat n$ for all $n$ and $\chat n\ge 0$
for all $n$, then for every $K$,
\[
(D'-X')\cdot\sumto \chat\,K\;+\;D'\cdot\chat\,K\;\le\;D'\cdot\chat\,0 .
\]
In particular $\sumto\chat\,K\le \chat\,0\cdot D'/(D'-X')$. The dual statement (persistence)
holds verbatim with inequalities reversed: if $D'\cdot\chat(n+1)\ge X'\cdot\chat n$ then
$(D'-X')\sumto\chat\,K+D'\cdot\chat\,K\ge D'\cdot\chat\,0$.
\end{lemma}
\begin{proof}
Identical induction to Lemma~\ref{lem:envelope} in the $(D',X')$ scale; the dual follows
by reversing the one inequality $(D'-X')\,\chat\,K+D'\,\chat\,(K+1)\ge D'\,\chat\,K$.
The anchor instances of Part~XI put $(D',X')=(269906323,250406323)$, $(742318937,703318937)$,
$(463042993,443542993)$ --- the exact critical ratios.
\end{proof}

\begin{proposition}[monotone two-channel potential; \texttt{phiMono}, \texttt{phiMono1..3}, \texttt{phiMonoAll}]\label{prop:potential}
If $D\cdot\dhat(n+1)\le Q\cdot\dhat n+D\cdot\chat n$, $D\cdot\chat(n+1)\le X\cdot\chat n$
with $0<X<D$, and $\dhat n,\chat n\ge 0$ for all $n$, then for every $K$, with $w=D-X$,
\[
(D-Q)\,w\cdot\sumto\dhat\,K\;+\;w\,D\cdot\dhat\,K\;+\;D^2\cdot\chat\,K
\;\le\;w\,D\cdot\dhat\,0\;+\;D^2\cdot\chat\,0 .
\]
\end{proposition}
\begin{proof}[Complete proof]
Define $\Phi_K=(D-Q)\,w\sumto\dhat\,K+w\,D\,\dhat\,K+D^2\chat\,K$;
the claim is $\Phi_K\le\Phi_0$. It suffices to show $\Phi_{K+1}\le\Phi_K$ and to note that
$(D-Q)\,w\sumto\dhat\,K\le\Phi_K$ (the other two summands are nonnegative). Now
\begin{align*}
\Phi_{K+1}-(D-Q)\,w\sumto\dhat\,K
&=(D-Q)\,w\,\dhat\,K+w\,D\,\dhat\,(K+1)+D^2\,\chat\,(K+1)\\
&\le(D-Q)\,w\,\dhat\,K+w\,\bigl(Q\,\dhat\,K+D\,\chat\,K\bigr)+X\,D\,\chat\,K\\
&=\bigl((D-Q)+Q\bigr)\,w\,\dhat\,K+D\,\chat\,K\,\bigl(w+X\bigr)\\
&=w\,D\,\dhat\,K+D^2\,\chat\,K ,
\end{align*}
using successively the coupled step inequality at $n=K$, the decay inequality at $n=K$,
the cancellation $(D-Q)+Q=D$, and the weight identity $w+X=(D-X)+X=D$. Hence
$\Phi_{K+1}\le (D-Q)\,w\sumto\dhat\,K+w\,D\,\dhat\,K+D^2\,\chat\,K=\Phi_K$.
\end{proof}

\begin{corollary}[drift bounds; \texttt{driftBound}, \texttt{driftBound1..3}, \texttt{controlTheorem}]\label{cor:bounds}
At each anchor the potential inequality plus caps yields $\sumto\dhat\,K\le$ the head:
V0 global: $X=22500000$ ($w=7500000$), head $1542000$;
anchor~1: $X=27832580$ ($w=2167420$), head $1541354$;
anchor~2: $X=28423857$ ($w=1576143$), head $1744063$;
anchor~3: $X=28736617$ ($w=1263383$), head $1928014$;
controller (all error channels adversarial at once, $\mu\le 3/4$):
$\tau\,0\le 39174$, head $1542000$.
\end{corollary}
\begin{proof}
From Proposition~\ref{prop:potential}, $(D-Q)\,w\sumto\dhat\,K\le w\,D\cdot 10^6+D^2\cdot 39000$
(the caps), so the numeric certificate
\[
w\,D\cdot 10^6+D^2\cdot 39000\;\le\;M\cdot\bigl((D-Q)\,w\bigr)
\]
--- verified as an exact integer multiplication at each anchor (e.g.\ anchor 1:
$2167420\cdot 30000000\cdot 1000000+30000000^2\cdot 39000\le 1541354\cdot 29970000\cdot 2167420$)
--- gives the head. Nonnegativity of $\dhat\,K,\chat\,K$ was used to discard the boundary terms.
\end{proof}

\begin{proposition}[the $1000/999$ accounting identity; \texttt{telescope}, \texttt{sumToTelescope}]\label{prop:telescope}
Under the \emph{exact} coupling recursion \eqref{eq:coupling}, for every $K$:
\begin{equation}\label{eq:telescope}
29970000\cdot\sumto\dhat\,K\;+\;30000000\cdot\dhat\,K
\;=\;30000000\cdot\bigl(\dhat\,0+\sumto\chat\,K\bigr).
\end{equation}
Equivalently, after dividing by $30000$:
$999\cdot\sumto\dhat\,K+1000\,\dhat\,K=1000\,(\dhat\,0+\sumto\chat\,K)$ --- the response
prefix is exactly $1000/999$ times (launch $+$ released source), minus the always-positive
boundary correction $\dhat\,K/999$.
\end{proposition}
\begin{proof}[Complete proof]
Induction on $K$. Base $K=0$: both sides are $30000000\,\dhat\,0$. Step: using
$\sumto\dhat\,(K+1)=\sumto\dhat\,K+\dhat\,K$ and the recursion
$30000000\,\dhat\,(K+1)=30000\,\dhat\,K+30000000\,\chat\,K$,
\begin{align*}
29970000\,\sumto\dhat\,(K+1)+30000000\,\dhat\,(K+1)
&=29970000\,\sumto\dhat\,K+29970000\,\dhat\,K\\
&\qquad+30000\,\dhat\,K+30000000\,\chat\,K\\
&=29970000\,\sumto\dhat\,K+30000000\,\dhat\,K+30000000\,\chat\,K\\
&\overset{\text{IH}}{=}30000000\,(\dhat\,0+\sumto\chat\,K)+30000000\,\chat\,K\\
&=30000000\,\bigl(\dhat\,0+\sumto\chat\,(K+1)\bigr),
\end{align*}
where $29970000+30000=30000000$ performed the absorption.
\end{proof}

\begin{proposition}[resolved certificate identity]\label{prop:cert}
Under \eqref{eq:coupling}, for every $K\ge 1$:
\begin{equation}\label{eq:cert}
999\cdot 1000^{K-1}\cdot\sumto\dhat\,K
=(1000^{K}-1)\,\dhat\,0\;+\;\sum_{j<K}\chat\,j\;\bigl(1000^{K}-1000^{\,j+1}\bigr),
\end{equation}
and every source weight $1000^{K}-1000^{\,j+1}$ on the right is a \emph{nonnegative} integer.
\end{proposition}
\begin{proof}
Divide \eqref{eq:telescope} by $30000$ and multiply by $1000^{K-1}$:
\begin{equation}\label{eq:mid}
999\cdot 1000^{K-1}\sumto\dhat\,K+1000^{K}\,\dhat\,K
=1000^{K}\,\dhat\,0+1000^{K}\sumto\chat\,K .
\end{equation}
It remains to expand $1000^{K}\,\dhat\,K$. Iterating \eqref{eq:coupling} in the form
$1000\,\dhat\,(j+1)=\dhat\,j+1000\,\chat\,j$ downward from $j=K-1$ to $j=0$ and
unrolling --- each turn of the unrolling moves one power of $1000$ off $\dhat$ and onto
$\chat$:
\[
1000^{K}\,\dhat\,K=1000^{K-1}\bigl(\dhat\,(K-1)+1000\,\chat\,(K-1)\bigr)
=\cdots=\dhat\,0+\sum_{j<K}1000^{\,j+1}\,\chat\,j .
\] Substituting this expansion of $1000^{K}\,\dhat\,K$ into \eqref{eq:mid} and
collecting the $\dhat\,0$ and $\chat\,j$ terms yields \eqref{eq:cert}; the weight
$1000^{K}-1000^{\,j+1}$ is nonnegative precisely when $j+1\le K$.
\end{proof}

\begin{remark}
This positivity is the entire \emph{mechanism} of the no-side: old sources weigh on the
prefix sum with large positive coefficients, so lower bounds on \emph{every}
$\chat\,j$ (never upper bounds) force $\sumto\dhat\,K$ upward. This is what lets the
squeeze of Part~XII run with no nonnegativity hypotheses at all.
\end{remark}

\section{The exact critical point (Part XI): detailed proof}\label{sec:critical}

\begin{theorem}[\texttt{criticalNever1..3}]\label{thm:critical}
Fix an anchor with head $M$ and let $\mu^\ast=c/d$ (lowest terms) be its exact critical
ratio, i.e.
\begin{equation}\label{eq:boundary}
(999\,M-10^{9})\,(d-c)=\chat_0\cdot 1000\cdot d,
\qquad \chat_0=39000,\quad \dhat_0^{\mathrm{cap}}=10^{6}.
\end{equation}
Under the exact coupling \eqref{eq:coupling}, the launch cap $\dhat\,0\le 10^{6}$, the
source cap $\chat\,0\le 39000$, the contraction bound
$d\cdot\chat(n+1)\le c\cdot\chat\,n$, and nonnegativity of both channels, the head is
never breached at any window: $\sumto\dhat\,K\le M$ for \emph{all} $K$.
\end{theorem}
\begin{proof}[Complete proof (anchor 1; anchors 2 and 3 verbatim)]
Instance: $(M,c,d)=(1541354,\,250406323,\,269906323)$; note $d-c=19500000$.
\emph{Step 1 (accounting).} From Proposition~\ref{prop:telescope} divided by $30000$:
\[
999\,\sigma+1000\,\dhat\,K=1000\,(\dhat\,0+\Sigma_c),\qquad
\sigma:=\sumto\dhat\,K,\quad\Sigma_c:=\sumto\chat\,K,
\]
so, since $\dhat\,K\ge 0$,
$999\,\sigma\le 1000\,(\dhat\,0+\Sigma_c)\le 1000\,(10^{6}+\Sigma_c)$.
\emph{Step 2 (source potential).} Lemma~\ref{lem:source} at $(D',X')=(269906323,250406323)$:
\[
19500000\,\Sigma_c\le 269906323\,(\chat\,0-\chat\,K)\le 269906323\cdot 39000 .
\]
\emph{Step 3 (combination).} From Step 2, $\Sigma_c\le 269906323\cdot 39000/19500000$.
Feeding this into Step 1 and multiplying through by $19500000$:
\[
999\cdot 19500000\,\sigma
\;\le\;1000\,\bigl(19500000\cdot 10^{6}+269906323\cdot 39000\bigr).
\]
The saturating identity (an exact integer multiplication --- the boundary identity in
weight form)
\[
29970000\cdot 19500000\cdot 1541354
=30000000\cdot\bigl(19500000\cdot 10^{6}+269906323\cdot 39000\bigr)
\]
turns the right side into exactly $999\cdot 19500000\cdot 1541354$, i.e.\
$\sigma\le 1541354$. In Lean the whole combination is one \texttt{omega} call over the two
potential equalities plus nonnegativity --- exact, no floats, no slack absorbed.
\end{proof}

\begin{remark}[where $\mu^\ast$ comes from]\label{rem:mustar}
A trajectory contracting at exactly $\mu^\ast$ would release, in the limit, the total
source $T^{\ast}=(999/1000)\,M-\dhat_0^{\mathrm{cap}}$ (inverting
$\sigma\le(1000/999)(\dhat_0+T)$ at equality), so the critical contraction ratio is
\[
\mu^{\ast}=1-\frac{\chat_0}{T^{\ast}}=1-\frac{1000\,\chat_0}{999\,M-10^{9}},
\]
and \eqref{eq:boundary} is exactly this statement after reduction to lowest terms
(anchor 1: $999\cdot 1541354-10^{9}=539812646$ and
$539812646\cdot 19500000=39000\cdot 1000\cdot 269906323$). The theorems show the head is
\emph{saturated} in the limit but \emph{never breached} at any finite window --- the
boundary is sharp, not merely safe.
\end{remark}

\subsection{Part XII: attainment at the exact critical ratio}\label{sec:squeeze}

\begin{theorem}[\texttt{noSideForcedSharp1..3}, \texttt{sharpForced1..3}]\label{thm:squeeze}
Fix an anchor $(M,a,b,A,K)$ as in Section~\ref{sec:reduction} ($A=(30000000/b)\,a$, so
$A/30000000=a/b$ exactly). Suppose:
\begin{enumerate}
\item[(H1)] exact coupling \eqref{eq:coupling};
\item[(H2)] floor law at the exact critical ratio: $30000000\cdot\chat(n+1)\ge A\cdot\chat\,n$ for all $n$;
\item[(H3)] launch at cap: $\dhat\,0\ge 10^{6}$; \quad (H4) floor at cap: $\chat\,0\ge 39000$.
\end{enumerate}
Then every admissible integer trajectory is forced above the head by the certified window:
\[
\sumto\dhat\,K\;\ge\;M+1 .
\]
\end{theorem}

\begin{proof}[Complete proof]
Define the three exact integer chains
\[
L_0=39000,\quad L_n=\Bigl\lceil\frac{a\,L_{n-1}}{b}\Bigr\rceil;
\qquad
XB_0=10^{6},\quad XB_n=XB_{n-1}+1000^{\,n}\,L_{n-1};
\qquad
CUM_0=0,\quad CUM_n=1000\,CUM_{n-1}+999\,XB_{n-1},
\]
and claim the three chain facts for $n=0,\dots,K$:
\begin{enumerate}
\item[$\mathrm{(i)}$] $\chat\,n\ge L_n$;\qquad
$\mathrm{(ii)}$\ $1000^{\,n}\,\dhat\,n\ge XB_n$;\qquad
$\mathrm{(iii)}$\ $999\cdot 1000^{\,n-1}\,\sumto\dhat\,n\ge CUM_n$\ (for $n\ge 1$).
\end{enumerate}
\emph{Chain (i).} Induction. Base: $\chat\,0\ge 39000=L_0$. Step: H2 gives
$30000000\,\chat\,(n+1)\ge A\,\chat\,n$; dividing by the exact positive factor
$30000000/b$ gives $b\,\chat\,(n+1)\ge a\,\chat\,n\ge a\,L_n$; the emitted literal check
$b\,L_{n+1}\ge a\,L_n$ (true by the ceiling definition of $L_{n+1}$) then forces
$\chat\,(n+1)\ge L_{n+1}$.

\emph{Chain (ii).} From \eqref{eq:coupling}, $1000\,\dhat\,(n+1)=\dhat\,n+1000\,\chat\,n$;
multiplying by $1000^{\,n}$ and applying chains (i) and (ii) at $n$:
\[
1000^{\,n+1}\,\dhat\,(n+1)=1000^{\,n}\,\dhat\,n+1000^{\,n+1}\,\chat\,n
\ge XB_n+1000^{\,n+1}\,L_n=XB_{n+1}.
\]

\emph{Chain (iii).} Weight the step equation $\sumto\dhat\,(n+1)=\sumto\dhat\,n+\dhat\,n$
by $999\cdot 1000^{\,n-1}$ and apply chains (ii) and (iii) at $n$:
\[
999\,1000^{\,n}\,\sumto\dhat\,(n+1)
=1000\cdot\bigl(999\,1000^{\,n-1}\sumto\dhat\,n\bigr)+999\cdot\bigl(1000^{\,n}\,\dhat\,n\bigr)
\ge 1000\,CUM_n+999\,XB_n=CUM_{n+1}.
\]

\emph{Close-out.} Chain (iii) at $n=K$ reads $LSC\cdot\sigma\ge CUM_K$ with
$LSC=999\cdot 1000^{K-1}$, the left scale of identity \eqref{eq:cert}. Chains (i) and
Proposition~\ref{prop:cert} then give
\[
LSC\,\sigma\;\overset{\text{(iii)}}{\ge}\;CUM_K\;=\;
(1000^{K}-1)\,\dhat\,0+\sum_{j<K}\chat\,j\,(1000^{K}-1000^{\,j+1})
\;\overset{\text{(i), (H3)}}{\ge}\;
(1000^{K}-1)\,10^{6}+\sum_{j<K}L_j\,(1000^{K}-1000^{\,j+1}),
\]
the last expression being the anchor's \emph{certificate value} $\mathrm{cert}_K$, an
exact integer; the certified crossing is the exact check $\mathrm{cert}_K\ge LSC\,(M+1)$.
At the anchors the one-window slacks $\mathrm{cert}_K-(M+1)$ and $\mathrm{cert}_{K-1}-(M+1)$
straddle zero, so $K$ is the exact crossing length. In Lean the identity \eqref{eq:cert}
is never stated closed-form: it is \emph{consumed} window by window --- each emitted
window proves its links of (i)--(iii) with literal numerals and immediately
\texttt{clear}s the consumed ones, so every \texttt{omega} call sees a bounded sliding
slice; the last window ends at $LSC\,\sigma\ge LSC\,(M+1)$ and one final \texttt{omega}
divides by the positive constant $LSC$.
\end{proof}

\section{Constructed singularities and controlled errors (Part XIII): detailed proof}\label{sec:witness}

Theorem~\ref{thm:squeeze} \emph{assumes} its four hypotheses. Part~XIII discharges them:
for each anchor it \emph{constructs} an integer trajectory satisfying H1--H4 exactly.

\subsection{The construction}
For each anchor, a state row $(\mathrm{bnd},\mathrm{tval},\chat,\mathrm{dval})\in\mathbb{Z}^4$
evolves by the exact integer recursion, launched at the initial row
$(39000,\,1000,\,\patch\,39000\,(0-1000),\,10^{6})$:
\begin{align}
\mathrm{bnd}'&=\frac{a\,\chat+(b-1)}{b} &&\text{(ceiling iterate of the floor law)},\label{eq:bnd}\\
\mathrm{tval}'&=\frac{\mathrm{tval}+\chat}{1000} &&\text{(integer part of the growth)},\label{eq:tval}\\
\chat'&=\patch\;\mathrm{bnd}'\;(0-\mathrm{tval}') &&\text{(the constructed error)},\label{eq:chat}\\
\mathrm{dval}'&=\mathrm{tval}+\chat .\label{eq:dval}
\end{align}

\begin{lemma}[the patch; \texttt{patchDeltaRange}, \texttt{patchGe}, \texttt{patchCongr}, \texttt{modSumZero}]\label{lem:patch}
With $\patch\;b\;p=b+\bigl((p-b)\bmod 1000+1000\bigr)\bmod 1000$:
\begin{enumerate}
\item[(a)] $0\le\patch\,b\,p-b\le 999$ \emph{(the lift is the entire error, bounded by $999$)};
\item[(b)] $b\le\patch\,b\,p$;
\item[(c)] $(\patch\,b\,p)\bmod 1000=p\bmod 1000$ \emph{(residue transfer)};
\item[(d)] $(t+(0-t)\bmod 1000)\bmod 1000=0$ for every $t$ \emph{(pinning realization)}.
\end{enumerate}
So $\patch\,b\,p$ is the least integer $\ge b$ congruent to $p\bmod 1000$.
\end{lemma}
\begin{proof}
(a), (b): $(p-b)\bmod 1000$ is the Euclidean remainder, lying in $[0,999]$ for every
integer $p-b$ (machine fact \texttt{modRange}, split into two separate single-atom goals
--- see the prover note below); adding $b$ gives both claims. (c):
$b+((p-b)\bmod 1000)\equiv b+(p-b)=p\pmod{1000}$. (d): immediate.
\end{proof}

\subsection{The invariants and the law}

\begin{proposition}[\texttt{invariantsW1..3}]\label{prop:inv}
For the constructed trajectory \eqref{eq:bnd}--\eqref{eq:dval}, at every $n$:
\begin{enumerate}
\item[(1)] $\mathrm{bnd}\,(n+1)=\dfrac{a\,\chat\,n+(b-1)}{b}$ \emph{(law ceiling)};
\item[(2)] $\mathrm{tval}\,(n+1)=\dfrac{\mathrm{tval}\,n+\chat\,n}{1000}$ \emph{(exact division)};
\item[(3)] $\chat\,(n+1)=\patch\,(\mathrm{bnd}\,(n+1))\,(0-\mathrm{tval}\,(n+1))$ \emph{(residue pinning)};
\item[(4)] $\mathrm{dval}\,n=1000\cdot\mathrm{tval}\,n$ \emph{(the $1000$-adic filtration)};
\item[(5)] $(\mathrm{tval}\,(n+1)+\chat\,(n+1))\bmod 1000=0$ \emph{(next division exact)};
\item[(6)] positivity: $\mathrm{bnd}\,n\ge 1$, $\mathrm{tval}\,n\ge 0$, $\chat\,n\ge 1$.
\end{enumerate}
\end{proposition}
\begin{proof}[Complete proof (induction on $n$)]
\emph{Base $n=0$.} The launch row gives (1), (2), (4), (6) by evaluation
($\mathrm{dval}\,0=10^{6}=1000\cdot 1000$). For (3): $\patch\,39000\,(0-1000)=39000$
because $(-40000)\bmod 1000=0$; the machine proof evaluates \texttt{patch} by
\texttt{unfold patch; omega} on the literal nested mods. (5):
$(1000+39000)\bmod 1000=0$.
\emph{Step $n\mapsto n+1$.} Assume (1)--(6) at $n$. \emph{(2) is exact}: by (5) at $n$,
$\mathrm{tval}\,n+\chat\,n\equiv 0\pmod{1000}$, so the floor division \eqref{eq:tval}
has no remainder --- write $q$ for the quotient. \emph{(3)} is the definition
\eqref{eq:chat}. \emph{(4)}: $\mathrm{dval}\,(n+1)=\mathrm{tval}\,n+\chat\,n=1000\,q
=1000\cdot\mathrm{tval}\,(n+1)$. \emph{(5)}: by Lemma~\ref{lem:patch}(c) applied at
$(\mathrm{bnd}\,(n+1),\,0-\mathrm{tval}\,(n+1))$ (the machine call \texttt{patchCongr})
and Lemma~\ref{lem:patch}(d) at $t=\mathrm{tval}\,(n+1)=q$:
$\bigl(\mathrm{tval}\,(n+1)+\chat\,(n+1)\bigr)\bmod 1000
=\bigl(q+(0-q)\bigr)\bmod 1000=0$. \emph{(1)} is the definition \eqref{eq:bnd}.
\emph{(6)} is preserved: by (1) at $n+1$ with $\chat\,n\ge 1$ and (4) at $n$ the numerator
$a\,\chat\,n+(b-1)$ stays positive, and the patch keeps
$\chat\,(n+1)\ge\mathrm{bnd}\,(n+1)\ge 1$ (Lemma~\ref{lem:patch}(b)).
In Lean, the zero case is \emph{manualized} (the \texttt{rfl}-evaluations
\texttt{e1..e4} with the rewrite chains \texttt{hpin0}/\texttt{hpinz}), because
\texttt{omega} does not fire function congruence between
$\patch\langle\text{atom args}\rangle$ and $\patch\langle\text{literal args}\rangle$;
the succ case assembles the six components from the displayed algebra.
\end{proof}

\begin{proposition}[the geometric floor law, attained; \texttt{lawW1..3}]\label{prop:law}
For the constructed trajectory: $b\cdot\chat\,(n+1)\ge a\cdot\chat\,n$ for every $n$;
equivalently on the coupling scale $30000000\cdot\chat\,(n+1)\ge A\cdot\chat\,n$, which is
exactly H2 of Theorem~\ref{thm:squeeze}.
\end{proposition}
\begin{proof}
Write $\mathrm{num}=a\,\chat\,n+(b-1)$ (nonnegative by positivity). Then
$\mathrm{bnd}\,(n+1)=\mathrm{num}/b$ has Euclidean remainder
$r=\mathrm{num}\bmod b\in[0,b-1]$, so
$b\cdot\mathrm{bnd}\,(n+1)=\mathrm{num}-r\ge\mathrm{num}-(b-1)=a\,\chat\,n$.
By Lemma~\ref{lem:patch}(b), $\chat\,(n+1)=\patch\,\mathrm{bnd}\,(n+1)\,(\cdot)\ge
\mathrm{bnd}\,(n+1)$. Multiplying $b\,\chat\,(n+1)\ge a\,\chat\,n$ by the exact positive
factor $30000000/b$ gives the coupling scale with $A=(30000000/b)\,a$.
(Machine form: \texttt{unfold patch} inside the \texttt{have} sources, then
\texttt{omega} over \texttt{patchGe}.)
\end{proof}

\begin{theorem}[\texttt{constructedWitness1..3}]\label{thm:witness}
For each anchor there \emph{exist} integer sequences $d,c:\mathbb{N}\to\mathbb{Z}$ with
\[
\text{\eqref{eq:coupling} exact}\ \wedge\ 30000000\,c(n+1)\ge A\,c\,n\
\wedge\ d\,0\ge 10^{6}\ \wedge\ c\,0\ge 39000\ \wedge\ \sumto d\,K\ge M+1 .
\]
\end{theorem}
\begin{proof}
Take $d\,n=\mathrm{dval}\,n$, $c\,n=\chat\,n$ from the construction. The coupling is exact:
\[
30000000\,\mathrm{dval}\,(n+1)=30000000\,(\mathrm{tval}\,n+\chat\,n)
=30000\,(1000\,\mathrm{tval}\,n)+30000000\,\chat\,n
=30000\,\mathrm{dval}\,n+30000000\,\chat\,n,
\]
using Proposition~\ref{prop:inv}(4) twice. The caps and H2 are
Propositions~\ref{prop:inv} and~\ref{prop:law}; the crossing is Theorem~\ref{thm:squeeze}
applied to \emph{these} witnesses --- the seven-component constructor in Lean ends in
\texttt{sharpForced$i$}, closing the loop with \emph{no hypothesis assumed}. All Part~XIII
exports are choice-free: $[\mathrm{propext},\mathrm{Quot.sound}]$.
\end{proof}

\section{The adversarial error bracket (Part XIV): detailed proof}\label{sec:bracket}

Part~XIV asks the control question directly: \emph{how much per-window adversarial error
can the mechanism absorb before the head-crossing fails --- and conversely, when does an
error already admit a counterexample?} The answer is a machine-checked two-sided bracket
per anchor: a yes-side tolerance $\mathsf{Em}$ over \emph{all} trajectories, and a no-side
frontier $\mathsf{Ens}$ with a \emph{constructed} counterexample, $\mathsf{Em}<\mathsf{Ens}$.

\subsection{Yes side: forcing under error}

\begin{theorem}[\texttt{forcingUnderError1..3}]\label{thm:forcing}
Fix an anchor $(M,a,b,A,K)$ with $\mathrm{mult}=30000000/b$, and let $\mathsf{Em}$ be the
anchor's certificate tolerance (computed below). Suppose:
\begin{enumerate}
\item[(E1)] exact coupling \eqref{eq:coupling};
\item[(E2)] \emph{perturbed} floor law: for all $n$,
\begin{equation}\label{eq:pertlaw}
b\cdot\chat(n+1)\;\ge\;a\cdot\chat\,n-\mathsf{Em},
\qquad\text{equivalently (exactly)}\qquad
30000000\,\chat\,(n+1)+\mathrm{mult}\cdot\mathsf{Em}\;\ge\;A\,\chat\,n;
\end{equation}
\item[(E3)] $\dhat\,0\ge 10^{6}$; \qquad (E4) $\chat\,0\ge 39000$.
\end{enumerate}
Then every admissible integer trajectory crosses the head by the certified window:
$\sumto\dhat\,K\ge M+1$.
\end{theorem}

\begin{remark}
The equivalence displayed in (E2) is exact, not approximate: since
$30000000=\mathrm{mult}\cdot b$ and $A=\mathrm{mult}\cdot a$ with $\mathrm{mult}>0$,
\[
\mathrm{mult}\,\bigl(b\,\chat\,(n+1)+\mathsf{Em}\bigr)\ge \mathrm{mult}\cdot a\,\chat\,n
\iff b\,\chat\,(n+1)+\mathsf{Em}\ge a\,\chat\,n
\]
--- the coupling-scale statement \emph{is} the b-scale statement multiplied through by the
known positive integer $\mathrm{mult}$; no rational rounding is introduced anywhere.
This fact ($\mathrm{F1}$) is what makes the whole bracket live at the integer level.
\end{remark}

\begin{proof}[Complete proof]
The proof is the sliding-window squeeze of Theorem~\ref{thm:squeeze} run on the
\emph{deficit floor lattice}
\[
L_0=39000,\qquad L_n=\Bigl\lceil\frac{a\,L_{n-1}-\mathsf{Em}}{b}\Bigr\rceil\quad(n\ge 1),
\]
with the response and prefix chains exactly as in Section~\ref{sec:squeeze} --- they
consume the floor chain only through the per-window bound $\chat\,n\ge L_n$, so the lattice
is interchangeably the clean one or the deficit one. Two facts carry each window:
\begin{enumerate}
\item[$\mathrm{(F1)}$] the scale equivalence of the remark above;
\item[$\mathrm{(F2)}$] \emph{lattice step.} The emitted literal check
$b\,L_n+\mathsf{Em}\ge a\,L_{n-1}$ (true by the ceil definition of $L_n$ in the presence
of the deficit) combines with (E2) at $n-1$ and the induction hypothesis
$\chat\,(n-1)\ge L_{n-1}$ as
\[
b\,\chat\,n+\mathsf{Em}\;\ge\;a\,\chat\,(n-1)\;\ge\;a\,L_{n-1}\;\ge\;b\,L_n+\mathsf{Em},
\]
whence $\chat\,n\ge L_n$.
\end{enumerate}
With $\chat\,n\ge L_n$ re-established at every window, chains (ii) and (iii) of
Theorem~\ref{thm:squeeze} run unchanged, and the close-out is the exact check, performed at
generation time,
\[
\mathrm{cert}(\mathsf{Em}):=(1000^{K}-1)\,10^{6}+\sum_{j<K}L_j\,(1000^{K}-1000^{\,j+1})
\;\ge\;LSC\,(M+1),\qquad LSC=999\cdot 1000^{K-1}.
\]
\emph{The tolerance is maximal}: the bisection in the design computation is boundary-exact
--- at deficit $\mathsf{Em}+1$ the same certificate value falls below $LSC\,(M+1)$, so the
raw floor chain no longer forces the crossing.
\end{proof}

\subsection{No side: constructed counterexamples under a strictly larger error}

\begin{proposition}[the pinning is not an extra hypothesis]\label{prop:forced}
Under the \emph{exact} coupling \eqref{eq:coupling}, every integer trajectory satisfies
$1000\mid\dhat\,n$ for all $n$; and, writing $\dhat\,n=1000\,s\,n$,
\[
\chat\,n\;\equiv\;-\,s\,n\;=\;-\,\dhat\,n/1000\pmod{1000}\qquad\text{for all }n .
\]
\end{proposition}
\begin{proof}
Reduce \eqref{eq:coupling} mod $1000$: the left side is $\equiv 0$ and
$30000000\,\chat\,n\equiv 0$, so $\dhat\,n\equiv 0\pmod{1000}$ --- at \emph{every} $n$,
with no hypothesis on $\dhat\,0$ whatsoever. Write $\dhat\,n=1000\,s\,n$; substituting into the exact step form
of \eqref{eq:coupling} obtained by dividing it by $30$,
$1000\,\dhat\,(n+1)=\dhat\,n+1000\,\chat\,n=1000(\,s\,n+\chat\,n)$,
gives $\dhat\,(n+1)=s\,n+\chat\,n$. Since also $1000\mid\dhat\,(n+1)$ (same argument one window
later), $s\,n+\chat\,n\equiv 0\pmod{1000}$, i.e.\ $\chat\,n\equiv -s\,n$.
\end{proof}

\texttt{patch} is precisely the canonical operation realizing the forced class while
preserving the floor law (Lemma~\ref{lem:patch}). This is why the counterexamples below

\subsection{The shape of the bracket}

The yes-side tolerance (hundreds to thousands on the b-scale) is far below the no-side
frontier (tens of millions), and this is structural, not slack: the yes side quantifies
over \emph{all} trajectories, whose worst case is the raw ceil-iterate chain, while the
constructed trajectory must live inside the forced residue classes of
Proposition~\ref{prop:forced} --- and those classes, threaded through the coupling, keep
the source floor elevated far above the raw lattice (that elevation is exactly why the
witness sums, $1.5$--$1.9$ million, sit so close to the heads). Between $\mathsf{Em}$ and
$\mathsf{Ens}$ lies the admissibility-stratified unknown, which narrows as more trajectory
structure is available to the adversary's quantifier. The honest statement of Part~XIV is
the pair of theorems per anchor: Theorem~\ref{thm:forcing} (deficit $\le\mathsf{Em}$
forces the crossing for everyone) and Theorem~\ref{thm:perturbed} (deficit
$\ge\mathsf{Ens}$ already admits a counterexample), both boundary-exact at their ends.
